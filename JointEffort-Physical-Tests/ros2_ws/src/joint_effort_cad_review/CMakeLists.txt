cmake_minimum_required(VERSION 3.16)
project(joint_effort_cad_review)
find_package(ament_cmake REQUIRED)
install(DIRECTORY launch urdf config DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/review_pose.py scripts/review_sliders.py DESTINATION lib/${PROJECT_NAME})
find_package(rviz_common REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
find_package(rclcpp REQUIRED)
add_executable(cad_rviz src/cad_rviz.cpp)
target_compile_features(cad_rviz PRIVATE cxx_std_17)
target_link_libraries(cad_rviz rviz_common::rviz_common Qt5::Widgets rclcpp::rclcpp)
install(TARGETS cad_rviz DESTINATION lib/${PROJECT_NAME})
ament_package()
