cmake_minimum_required(VERSION 3.16)
project(joint_effort_hardware)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
add_library(sts_bus SHARED src/sts_bus.cpp)
target_include_directories(sts_bus PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
add_library(iron_system SHARED src/iron_system.cpp)
add_executable(sts_inspect src/sts_inspect.cpp)
target_link_libraries(sts_inspect sts_bus)
target_link_libraries(iron_system sts_bus)
ament_target_dependencies(iron_system hardware_interface pluginlib rclcpp rclcpp_lifecycle)
pluginlib_export_plugin_description_file(hardware_interface hardware.xml)
ament_python_install_package(${PROJECT_NAME})
install(TARGETS sts_bus iron_system DESTINATION lib)
install(TARGETS sts_inspect DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/sts_commission DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_sts_bus test/test_sts_bus.cpp)
  target_link_libraries(test_sts_bus sts_bus pthread)
endif()
ament_package()
