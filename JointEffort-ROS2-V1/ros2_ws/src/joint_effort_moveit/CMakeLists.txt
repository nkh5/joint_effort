cmake_minimum_required(VERSION 3.16)
project(joint_effort_moveit)
find_package(ament_cmake REQUIRED)
find_package(moveit_ros_move_group REQUIRED)
add_executable(move_group_lifetime src/move_group_lifetime.cpp)
target_compile_features(move_group_lifetime PRIVATE cxx_std_17)
target_link_libraries(move_group_lifetime moveit_ros_move_group::moveit_move_group_capabilities_base)
install(TARGETS move_group_lifetime DESTINATION lib/${PROJECT_NAME})
add_library(joint_effort_no_execution_controller_manager SHARED src/no_execution_controller_manager.cpp)
target_compile_features(joint_effort_no_execution_controller_manager PRIVATE cxx_std_17)
target_link_libraries(joint_effort_no_execution_controller_manager moveit_ros_move_group::moveit_move_group_capabilities_base)
pluginlib_export_plugin_description_file(moveit_core controllers_plugin.xml)
install(TARGETS joint_effort_no_execution_controller_manager LIBRARY DESTINATION lib)
add_executable(mount_clearance src/mount_clearance.cpp)
target_compile_features(mount_clearance PRIVATE cxx_std_17)
target_link_libraries(mount_clearance moveit_ros_move_group::moveit_move_group_capabilities_base)
install(TARGETS mount_clearance DESTINATION lib/${PROJECT_NAME})
add_executable(iron_collision_audit src/iron_collision_audit.cpp)
target_compile_features(iron_collision_audit PRIVATE cxx_std_17)
target_link_libraries(iron_collision_audit moveit_ros_move_group::moveit_move_group_capabilities_base)
install(TARGETS iron_collision_audit DESTINATION lib/${PROJECT_NAME})
find_package(moveit_servo REQUIRED)
add_executable(servo_lifetime src/servo_lifetime.cpp)
target_compile_features(servo_lifetime PRIVATE cxx_std_17)
target_link_libraries(servo_lifetime moveit_servo::moveit_servo_lib_ros)
install(TARGETS servo_lifetime DESTINATION lib/${PROJECT_NAME})
ament_package()
